"reports" }} {{- $filters = $filters | append (images.Text (i18n "strings.articles" | upper) (dict "font" $sans "size" 22 "color" "#6e4242" "x" 82 "y" 134)) -}} {{ end }}{{ end -}} {{- $result = ($base | images.Filter $filters).Process "jpg q85" -}} {{- end -}} {{- return $result -}}
